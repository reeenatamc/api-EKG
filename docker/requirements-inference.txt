# What Open-ECG-Digitizer's own src/ actually imports at inference time, and nothing else.
#
# The digitizer's own requirements.txt (in its upstream repo) also pulls in ray[data,train,
# tune], tensorboard, transformers, jupyter, plotly and torch-audio -- its training and
# notebook tooling, none of which the digitize.py subprocess this service shells out to
# ever imports. ray itself is made optional by ecg-pipeline's patches/0001, so it is not
# listed below either. See the deployment report, section 7, for how this list was reached.
#
# Versions are pinned to the exact ones validated on the digitizer's own virtualenv (Intel
# macOS, Python 3.12) -- see README, "Environment", on why this stack is pinned rather than
# left to the resolver. torch and torchvision are installed separately, from the PyTorch CPU
# index (see Dockerfile), so the CUDA build is never what the resolver picks by accident.

torchvision==0.17.2
torch-tps==1.2.2

# opencv-python-headless, not opencv-python: no GUI bindings to link against, and no libGL
# to install into the image for them -- this service never displays an image, only reads
# and writes arrays.
opencv-python-headless==4.9.0.80

scikit-image==0.22.0
scikit-learn==1.4.2
networkx==3.2.1
yacs==0.1.8
